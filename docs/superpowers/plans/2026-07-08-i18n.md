# i18n 中英文多语言 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Chinese/English bilingual support with manual language switch button, language preference persisted to localStorage.

**Architecture:** New `i18n.ts` module with `$state`-based locale store, `t()` lookup function, and dynamic JSON import for language packs. Components replace hardcoded strings with `t('key')` calls. Language switch button in header.

**Tech Stack:** Svelte 5 runes, TypeScript, dynamic `import()`, JSON language packs, localStorage

## Global Constraints

- Zero new npm dependencies
- Language packs in `client/src/lib/locales/en.json` and `client/src/lib/locales/zh.json`
- `t(key)` fallback chain: current locale → en → key itself
- Language preference persisted to `localStorage` key `ontology-lang`
- JSON files loaded via dynamic `import()` (Vite handles this natively)
- Key naming convention: `namespace.key` (e.g. `app.title`, `stats.classes`)
- Variable interpolation via `{n}` placeholder + `.replace()`

let currentDict: Record<string, string> = { ...enFallback }

/** Current locale, persisted to localStorage. */
export let locale = $state<Locale>(loadLocale())

/** Look up a translated string by key. */
export function t(key: string): string {
  return currentDict[key] ?? enFallback[key] ?? key
}

/** Switch language. */
export async function setLocale(lang: Locale) {
  locale = lang
  saveLocale(lang)
  try {
    const mod = await import(`./locales/${lang}.json`) as { default: Record<string, string> }
    currentDict = { ...enFallback, ...mod.default }
  } catch {
    // JSON load failed — keep current dict (or fallback).
  }
}

/** Call once on app mount to load initial language pack. */
export async function initLocale() {
  await setLocale(locale)
}

// ---- helpers ----

function loadLocale(): Locale {
  try {
    const v = localStorage.getItem('ontology-lang')
    if (v === 'zh') return 'zh'
  } catch { /* localStorage unavailable */ }
  return 'en'
}

function saveLocale(lang: Locale) {
  try { localStorage.setItem('ontology-lang', lang) } catch { /* ignore */ }
}
